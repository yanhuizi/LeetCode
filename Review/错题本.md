int search(int* nums, int numsSize, int target) {
    //int* nums即传过来的nums这个数组第一个元素的地址
    //约等于nums[]
    int left=0;
    int right=numsSize-1;//定义的是左右边界的索引
   
    while(left<=right)
    {
        int mid=(left+right)/2;//中间元素的索引
         if (target<nums[mid])
        {
            right=mid-1;
        }
        else if (target>nums[mid])
        {
            left=mid+1;
        }
        else
        {
            return mid;
        }
       
    }
    return -1;

}

- 正确思路：用 i 扫描整个数组，用 k 记录保留下来的位置。遇到不等于 val 的数，就丢到 nums[k]，然后 k++。也就是把所有“不等于 val 的元素”从前往后重新放到数组前面。

int removeElement(int* nums, int numsSize, int val) {
    //循环判断数组的值是否为val，若为val则删除（？替换别的数值）并计数为k，否则保留。
    //循环完毕返回不等于val值的数量k的大小以及新的数组
    int k=0;
    for (int i = 0; i < numsSize; i++)
    {
        if (nums[i]!=val)
        {
            nums[k]=nums[i];
            k++;
        }
        
    }
    return k;
}

### 35 搜索插入位置

- 日期：2026-07-30
- 错误原因：
1.O(n)：一次减少一个或少量元素，比如普通遍历。O(log n)：一次减少一半元素，比如二分查找。
不过 O(log n) 通常有条件。比如二分查找要求数组已经有序。如果数组没排序，你不能随便二分，只能老老实实遍历，除非先排序或换别的数据结构;
2.只写了二分查找，但其实只差最后一步return left,因为二分查找没找到时，left 停下来的位置就是插入位置，此时while循环条件就不满足了;

- 正确思路：二分查找加上return left即可。

int searchInsert(int* nums, int numsSize, int target) {
    int left=0;
    int right=numsSize-1;

    while (left<=right)
    {
        int mid=(right+left)/2;
        if (nums[mid]<target){
            left=mid+1;
        }else if (nums[mid]>target){
            right=mid-1;
        }else{
            return mid;
        }
    }
    return left;
    
}
